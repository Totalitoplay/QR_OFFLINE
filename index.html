function processCSVData(text, showFeedback = false) { const delimiter = text.indexOf(';') > -1 ? ';' : ','; const lines = text.split(/\r?\n/); if (lines.length < 2) return; const headers = lines[0].split(delimiter).map(h => clean(h).toUpperCase()); const iQR = Math.max(headers.findIndex(h => h.includes('QR')), 4); const iOLT = Math.max(headers.findIndex(h => h.includes('OLT')), 0); const iF = Math.max(headers.findIndex(h => h === 'F'), 1); const iS = Math.max(headers.findIndex(h => h === 'S'), 2); const iP = Math.max(headers.findIndex(h => h === 'P'), 3); const iCuenta = Math.max(headers.findIndex(h => h.includes('CUENTA')), 5); const iPuertoN2 = headers.findIndex(h => h.includes('PUERTO N2') || h.includes('PUERTON2') || h === 'N2'); const iLat = Math.max(headers.findIndex(h => h.includes('LATITUD')), 7); const iLon = Math.max(headers.findIndex(h => h.includes('LONGITUD')), 8); const iCap = Math.max(headers.findIndex(h => h.includes('CAPACIDAD')), 9); const iUbicacion = headers.findIndex(h => h.includes('UBICACION') || h.includes('UBICACIÓN')); offlineData = {}; let loadedCount = 0; for (let i = 1; i < lines.length; i++) { const row = lines[i].split(delimiter); if (row.length < 5) continue; const qrRaw = clean(row[iQR]); const qr = qrRaw.toUpperCase().replace(/\s+/g, ""); if (!qr) continue; // Calcular capacidad de la caja (total de puertos) let totalPorts = 16; const capStr = clean(row[iCap]); if (capStr.includes(':')) { totalPorts = parseInt(capStr.split(':')[1], 10) || 16; } else if (parseInt(capStr, 10) > 0) { totalPorts = parseInt(capStr, 10); } if (!offlineData[qr]) { offlineData[qr] = { nombre: clean(row[iOLT]), frame: clean(row[iF]), slot: clean(row[iS]), puerto: clean(row[iP]), latitud: clean(row[iLat]), longitud: clean(row[iLon]), ubicacion: iUbicacion !== -1 ? clean(row[iUbicacion]) : "", totalPuertos: totalPorts, // Inicializamos el arreglo de cuentas con la capacidad total en nulo/vacío cuentas: new Array(totalPorts).fill(null) }; loadedCount++; } const cuenta = clean(row[iCuenta]); // Si existe cuenta, buscamos a qué número de Puerto N2 corresponde if (cuenta) { let numPuerto = 0; if (iPuertoN2 !== -1 && row[iPuertoN2]) { numPuerto = parseInt(clean(row[iPuertoN2]), 10); } // Si se definió un Puerto N2 válido dentro del rango, se asigna a ese puerto específico if (numPuerto > 0 && numPuerto <= offlineData[qr].totalPuertos) { offlineData[qr].cuentas[numPuerto - 1] = cuenta; } else { // Fallback: Si no tiene Puerto N2 asignado, se busca el primer puerto libre disponible const idxLibre = offlineData[qr].cuentas.findIndex(c => c === null); if (idxLibre !== -1) { offlineData[qr].cuentas[idxLibre] = cuenta; } } } } document.getElementById('setup-view').style.display = 'none'; document.getElementById('app-view').style.display = 'flex'; if (showFeedback) { alert(`Base de datos actualizada. Cajas indexadas: ${loadedCount}`); } } function renderOffline(data, q, isFound = true) { const bannerError = document.getElementById("not-found-banner"); if (!isFound) { data = { nombre: "DESCONOCIDO", frame: "—", slot: "—", puerto: "—", totalPuertos: 0, cuentas: [], latitud: "", longitud: "", ubicacion: "" }; bannerError.style.display = "block"; } else { bannerError.style.display = "none"; } document.getElementById("n").textContent = data.nombre || "NODO SIN NOMBRE"; document.getElementById("fsp-f").textContent = data.frame || "—"; document.getElementById("fsp-s").textContent = data.slot || "—"; document.getElementById("fsp-p").textContent = data.puerto || "—"; const total = data.totalPuertos || 0; // Contamos únicamente las posiciones que tienen una cuenta asignada const ocupados = data.cuentas.filter(c => c !== null && c !== "").length; const libres = Math.max(0, total - ocupados); document.getElementById("li").textContent = libres; document.getElementById("to").textContent = total; const badge = document.getElementById("avail-badge"); const pct = total ? libres / total : 0; if (!isFound || pct === 0) { badge.style.cssText = "background:rgba(248,113,113,0.12);color:#f87171;border:1px solid rgba(248,113,113,0.2)"; } else if (pct < 0.3) { badge.style.cssText = "background:rgba(251,191,36,0.12);color:#fbbf24;border:1px solid rgba(251,191,36,0.2)"; } else { badge.style.cssText = "background:var(--green-dim);color:var(--green);border:1px solid rgba(52,211,153,0.2)"; } document.getElementById("qr-label").textContent = q; dibujarQR(q); const lat = data.latitud, lon = data.longitud; const coordsEl = document.getElementById("coords"); const mapsLink = document.getElementById("mapsLink"); const ubicacionEl = document.getElementById("ubicacion"); if (!isFound) { ubicacionEl.textContent = "Sin datos de ubicación"; } else { ubicacionEl.textContent = data.ubicacion ? `Ubicación: ${data.ubicacion}` : "Ubicación: (No especificada)"; } if (lat && lon && isFound) { coordsEl.textContent = `${lat}, ${lon}`; mapsLink.href = `https://www.google.com/maps/search/?api=1&query=${lat},${lon}`; mapsLink.style.display = "inline-block"; } else { coordsEl.textContent = "Coordenadas no disponibles"; mapsLink.style.display = "none"; } document.getElementById("ports-summary").textContent = `${libres} libre${libres !== 1 ? "s" : ""} de ${total}`; const grid = document.getElementById("portsGrid"); const frag = document.createDocumentFragment(); for (let i = 1; i <= total; i++) { const cuenta = data.cuentas[i - 1]; const libre = !cuenta; const cell = document.createElement("div"); cell.className = "port-cell" + (libre ? " is-free" : ""); const lbl = document.createElement("div"); lbl.className = "port-num-label"; lbl.textContent = "Puerto"; const num = document.createElement("div"); num.className = "port-num"; num.textContent = i; const st = document.createElement("div"); st.className = "port-status " + (libre ? "free" : "occ"); st.textContent = libre ? "LIBRE" : cuenta; cell.append(lbl, num, st); frag.appendChild(cell); } grid.replaceChildren(frag); document.getElementById("result").style.display = "flex"; window.scrollTo({ top: 0, behavior: "smooth" }); }
